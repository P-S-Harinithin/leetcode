class Solution {
    public int maxArea(int[] h) {
        int max=0;
        for(int i=0;i<n;i++){
            for(int j=i+1;j<n;j++){
                int h1=Math.min(h[i],h[j]);
            }
        }
                int w=j-i;

                int a=h1*w;
                if(a>max){
                    max=a;
                }
    }
        return max;
        int n=h.length;
